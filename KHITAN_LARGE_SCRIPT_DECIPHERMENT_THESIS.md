def calculate_character_frequency(self, character, context='administrative'):
    """
    Calculate frequency for Khitan character based on social hierarchy
    """
    if character in self.character_frequencies:
        base_freq = self.character_frequencies[character]
    else:
        # Default to administrative level for unknown characters
        base_freq = 550.0
    
    # Apply context amplification
    context_factor = self.context_amplifiers.get(context, 1.0)
    final_freq = base_freq * context_factor
    
    return final_freq

def classify_social_level(self, frequency):
    """
    Classify social hierarchy level based on frequency
    """
    if frequency >= 880:
        return 'imperial', '皇帝'
    elif frequency >= 660:
        return 'noble', '贵族'
    elif frequency >= 440:
        return 'administrative', '官员'
    else:
        return 'temporal', '临时'

def analyze_inscription_hierarchy(self, characters, context='administrative'):
    """
    Analyze complete inscription for social hierarchy patterns
    """
    results = []
    total_frequency = 0
    
    for char in characters:
        freq = self.calculate_character_frequency(char, context)
        level, chinese = self.classify_social_level(freq)
        
        results.append({
            'character': char,
            'frequency': freq,
            'social_level': level,
            'chinese_term': chinese,
            'context': context
        })
        
        total_frequency += freq
    
    # Calculate average hierarchy level
    avg_frequency = total_frequency / len(characters)
    overall_level, overall_chinese = self.classify_social_level(avg_frequency)
    
    return {
        'character_analysis': results,
        'average_frequency': avg_frequency,
        'overall_social_level': overall_level,
        'overall_chinese_term': overall_chinese,
        'document_context': context
    }

def apply_mongolian_harmonics(self, base_frequency):
    """
    Apply Mongolian throat singing harmonic principles
    """
    # Find closest khoomei harmonic
    harmonics = list(self.khoomei_harmonics.values())
    closest_harmonic = min(harmonics, key=lambda x: abs(x - base_frequency))
    
    # Calculate harmonic ratio
    harmonic_ratio = base_frequency / closest_harmonic
    
    # Generate overtone series
    overtones = []

    return {
